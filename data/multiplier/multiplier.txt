INPUT in1<31:0>, in2<31:0>;
REGISTER R1<31:0>, R2<31:0>, R3<31:0>, RR<0:0>, R4<31:0>, S<0,0>;
OUTPUT R4, RR;
INITIAL Idle;

Idle: STATE, {}, CheckStart;
CheckStart: DECISION, {S}, 0:Idle, 1:LoadOperands;
LoadOperands: CONDITIONAL, {R1=in1, R2=in2, RR=0}, Setup;
Setup: STATE, {R3=0}, CompareOperands;
CompareOperands: DECISION, {R1>R2}, 0:Loop, 1:SwapOperands;
SwapOperands: CONDITIONAL, {R1=R2, R2=R1}, Loop;
Loop: STATE, {}, CheckDone;
CheckDone: DECISION, {OR R1}, 0:WriteResult, 1:Accumulate;
Accumulate: CONDITIONAL, {R1=R1-1, R3=R3+R2}, Loop;
WriteResult: CONDITIONAL, {R4=R3, RR=1}, Idle;
